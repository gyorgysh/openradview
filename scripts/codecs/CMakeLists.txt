cmake_minimum_required(VERSION 3.20)
project(OpenRadViewCodecChecks LANGUAGES C CXX)
if(NOT CMAKE_C_COMPILER_ID MATCHES "Clang" OR NOT CMAKE_CXX_COMPILER_ID MATCHES "Clang")
  message(FATAL_ERROR "This local sanitizer check requires Clang")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer -g)
add_link_options(-fsanitize=address,undefined)
get_filename_component(ORV_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(CHARLS_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(CHARLS_BUILD_FUZZ_TEST OFF CACHE BOOL "" FORCE)
set(CHARLS_BUILD_SAMPLES OFF CACHE BOOL "" FORCE)
set(CHARLS_INSTALL OFF CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
add_subdirectory("${ORV_ROOT}/vendor/charls-sys/charls" charls)
set(JP2 "${ORV_ROOT}/vendor/openjpeg-sys/vendor/src/lib/openjp2")
set(JP2_FILES thread bio cio dwt event ht_dec image invert j2k jp2 mct mqc openjpeg opj_clock pi t1 t2 tcd tgt function_list opj_malloc sparse_array)
foreach(name IN LISTS JP2_FILES)
  list(APPEND JP2_SOURCES "${JP2}/${name}.c")
endforeach()
add_library(openjp2 STATIC ${JP2_SOURCES})
target_include_directories(openjp2 PUBLIC "${JP2}" "${ORV_ROOT}/vendor/openjpeg-sys/config")
target_compile_definitions(openjp2 PUBLIC OPJ_STATIC MUTEX_stub OPJ_HAVE_POSIX_MEMALIGN=1)
add_executable(codec-checks native.cpp)
target_link_libraries(codec-checks PRIVATE charls openjp2)
